ตรวจสอบประวัติการเบิกจ่าย & Google Sheets

ตรวจสอบประวัติย้อนหลัง ดึงข้อมูลเดิมขึ้นมาดู ปริ้น หรือส่งออก Google Sheets

วิธีเชื่อม Google Sheets ของคุณเอง:

  1. สร้าง Google Sheets ใหม่ ไปที่เมนู ส่วนขยาย > Apps Script
  2. วางโค้ดด้านล่าง แล้วกด การทำให้ใช้งานได้ > รายการใหม่ > เว็บแอป (ทุกคนเข้าถึงได้)
  3. คัดลอก URL มาวางในช่องด้านบน ข้อมูลการเบิกจะถูกบันทึกเข้า Sheets อัตโนมัติ
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var d = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), d.reqNo, d.requesterName, d.empId, d.department, d.purpose, JSON.stringify(d.items)]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
